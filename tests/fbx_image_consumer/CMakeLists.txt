cmake_minimum_required(VERSION 3.24)
project(iiActorObjectFbxImageInstalledConsumer LANGUAGES CXX)

option(TEST_VISUALCODEC_COMPONENT "Test the separately installed VisualCodec component" OFF)
option(TEST_NSCHARACTER_CONSUMER "Test the NSCharacter borrowed view" OFF)
if(TEST_VISUALCODEC_COMPONENT)
    find_package(iiActorObject 0.2.1 CONFIG REQUIRED COMPONENTS VisualCodec)
else()
    find_package(iiActorObject 0.2.1 CONFIG REQUIRED)
endif()

add_executable(iiActorObjectFbxImageInstalledConsumer main.cpp)
target_compile_features(iiActorObjectFbxImageInstalledConsumer PRIVATE cxx_std_23)
set_target_properties(iiActorObjectFbxImageInstalledConsumer PROPERTIES CXX_EXTENSIONS OFF)
if(TEST_VISUALCODEC_COMPONENT)
    target_link_libraries(iiActorObjectFbxImageInstalledConsumer PRIVATE iiActorObject::VisualCodec)
    target_compile_definitions(iiActorObjectFbxImageInstalledConsumer PRIVATE TEST_VISUALCODEC_COMPONENT)
else()
    target_link_libraries(iiActorObjectFbxImageInstalledConsumer PRIVATE iiActorObject::iiActorObject)
endif()
if(TEST_NSCHARACTER_CONSUMER)
    target_compile_definitions(iiActorObjectFbxImageInstalledConsumer PRIVATE TEST_NSCHARACTER_CONSUMER)
endif()

set(FBX_IMAGE_FIXTURE "" CACHE FILEPATH "External-image FBX fixture")
set(FBX_COLOR_FIXTURE "" CACHE FILEPATH "Color PNG fixture")
set(FBX_NORMAL_FIXTURE "" CACHE FILEPATH "Normal PNG fixture")
enable_testing()
add_test(NAME iiActorObject.installed_fbx_image_consumer COMMAND iiActorObjectFbxImageInstalledConsumer
    "${FBX_IMAGE_FIXTURE}" "${FBX_COLOR_FIXTURE}" "${FBX_NORMAL_FIXTURE}")
if(WIN32)
    set_tests_properties(iiActorObject.installed_fbx_image_consumer PROPERTIES
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:iiActorObject::iiActorObject>")
endif()
