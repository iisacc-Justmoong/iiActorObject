cmake_minimum_required(VERSION 3.24)
project(iiActorObjectRigSamplesInstalledConsumer LANGUAGES CXX)
find_package(iiActorObject 0.2.1 CONFIG REQUIRED)
if(TARGET Qt6::Core)
    message(FATAL_ERROR "The installed core must not import Qt")
endif()
add_executable(iiActorObjectRigSamplesInstalled ../Object3DRigSamplesTest.cpp)
target_link_libraries(iiActorObjectRigSamplesInstalled PRIVATE iiActorObject::iiActorObject)
set_target_properties(iiActorObjectRigSamplesInstalled PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
foreach(case endpoints partial failures lifetime)
    add_test(NAME installed.rig_samples.${case} COMMAND iiActorObjectRigSamplesInstalled ${case})
endforeach()
add_test(NAME installed.rig_samples.formats COMMAND iiActorObjectRigSamplesInstalled formats "${CMAKE_CURRENT_BINARY_DIR}/formats")
set(IIACTOROBJECT_DNA_FIXTURE_DIRECTORY "" CACHE PATH "Independent DNA fixture directory")
if(IIACTOROBJECT_DNA_FIXTURE_DIRECTORY)
    add_test(NAME installed.rig_samples.dna COMMAND iiActorObjectRigSamplesInstalled dna "${IIACTOROBJECT_DNA_FIXTURE_DIRECTORY}")
endif()
if(WIN32)
    get_property(test_names DIRECTORY PROPERTY TESTS)
    set_tests_properties(${test_names} PROPERTIES ENVIRONMENT_MODIFICATION
        "PATH=path_list_prepend:$<TARGET_FILE_DIR:iiActorObject::iiActorObject>")
endif()
